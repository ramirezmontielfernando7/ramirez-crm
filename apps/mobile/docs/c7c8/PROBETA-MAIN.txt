const fetchMock = vi.fn(async (input: string | URL, init?: RequestInit) => {

  if (url.origin !== base || !url.pathname.startsWith("/api/dev/wa-mock/graph/"))
    throw new Error("Network forbidden in fixture");
  const path = url.pathname.slice("/api/dev/wa-mock/graph/".length).split("/");

});

type Handler = (req: Request, context: { params: Promise<{ id: string }> }) => Promise<Response>;
async function call(
  handler: Handler, session: keyof typeof ids | null,
  method: string, path: string, body?: object, id = "",
) {
  const headers = new Headers(session ? native[session] : undefined);
  if (body && !(body instanceof FormData)) headers.set("content-type", "application/json");
  ctx.headers = headers;
  const req = new NextRequest(base + path, { method, headers, body: body instanceof FormData ? body : body && JSON.stringify(body) });
  expect(middleware(req).status).toBe(200);
  return handler(req, { params: Promise.resolve({ id }) });
}

async function bodyOk(response: Response, status = 200) {
  expect(response.status, response.ok ? undefined : await response.clone().text()).toBe(status);
  return response.json();
}

it("media inherits browser CSRF, contact scope and a single wa-mock send; binary GET uses Bearer", async () => {
  const make = () => {
    const form = new FormData();
    form.set("file", new File(["%PDF-1.4\nFixture de prueba"],"fixture.pdf",{type:"application/pdf"}));
    return form;
  };
  const attempt = (actor: "owner" | "advisor", index: number) => {
    const id=chats[index];
    return call(sendMedia,actor,"POST","/api/conversations/"+id+"/messages/media",make(),id);
  };
  const before=getWaMockState().outbox.length;
  expect((await attempt("advisor",0)).status).toBe(404);
  expect((await attempt("owner",2)).status).toBe(404);
  expect(getWaMockState().outbox.length).toBe(before);
  const created=await bodyOk(await attempt("advisor",1),201);
  expect(getWaMockState().outbox.length).toBe(before+1);
  expect((getWaMockState().outbox.at(-1)?.body as { type: string }).type).toBe("document");
  const [row]=await db.select({assetId:schema.message.mediaAssetId}).from(schema.message).where(eq(schema.message.id,created.messageId));
  const assetId=row!.assetId!;
  const download=(actor:keyof typeof ids | null)=>call(
    (req)=>readMedia(req,{params:Promise.resolve({assetId})}),actor,"GET","/api/media/"+assetId
  );
  expect((await download(null)).status).toBe(401);
  expect((await download("foreign")).status).toBe(404);
  const response=await download("advisor");
  expect(response.status).toBe(200);
  expect(response.headers.get("content-type")).toBe("application/pdf");
  expect(await response.text()).toContain("%PDF-1.4");
  await db.update(schema.conversation).set({lastInboundAt:new Date(0)}).where(eq(schema.conversation.id,chats[1]));
  expect((await attempt("advisor",1)).status).toBe(409);
  expect(getWaMockState().outbox.length).toBe(before+1);
});

it("pipeline stage updates and results follow the web visibility",async()=>{
  const list=async(actor:"owner"|"advisor")=>(await bodyOk(await call(board,actor,"GET","/api/pipeline/board?take=50"))).leads.map((row:{id:string})=>row.id);
  expect(await list("advisor")).toEqual([leads[1]]);
  expect(new Set(await list("owner"))).toEqual(new Set(leads.slice(0,2)));
  const stagesA=stages.filter(r=>r.organizationId===orgs[0]);
  const apply=(actor:"owner"|"advisor",index:number,stageId:string,lossReason?:string)=>call(move,actor,"PATCH","/api/pipeline/leads/"+leads[index],{stageId,...(lossReason?{lossReason}:{})},leads[index]);
  expect((await apply("advisor",0,stagesA[1]!.id)).status).toBe(404);
  expect((await apply("owner",2,stagesA[1]!.id)).status).toBe(404);
  expect((await apply("advisor",1,"invalid_stage")).status).toBe(422);
  expect((await bodyOk(await apply("advisor",1,stagesA[1]!.id))).lead.stageId).toBe(stagesA[1]!.id);
  expect((await apply("advisor",1,stagesA[3]!.id)).status).toBe(422);
  expect((await bodyOk(await apply("advisor",1,stagesA[3]!.id,"otro"))).lead.stageId).toBe(stagesA[3]!.id);
  expect((await call(results,"advisor","GET","/api/analytics/sales")).status).toBe(403);
  for (const [actor,count] of [["owner",2],["foreign",1]] as const) {
    const snapshot=await bodyOk(await call(results,actor,"GET","/api/analytics/sales"));
    expect(snapshot.kpis.newLeads.current).toBe(count);
  }
  const own=(await bodyOk(await call(results,"owner","GET","/api/analytics/sales?userId="+ids.advisor))).kpis.newLeads.current;
  expect(own).toBe(1);
  expect((await call(teamResults,"advisor","GET","/api/analytics/team")).status).toBe(403);
});

it("AI uses the exact scoped PATCH while privileged listings keep 403 and campaign tenants keep 404",async()=>{
  const ai=(actor:"advisor"|"owner",index:number,body:object)=>call(patchConversation,actor,"PATCH","/api/conversations/"+chats[index],body,chats[index]);
  expect((await bodyOk(await ai("advisor",1,{aiEnabled:false}))).conversation.aiEnabled).toBe(false);
  expect((await bodyOk(await ai("advisor",1,{reactivate:true}))).conversation.aiEnabled).toBe(true);
  expect((await ai("advisor",0,{aiEnabled:false})).status).toBe(404);
  expect((await ai("owner",2,{reactivate:true})).status).toBe(404);
  expect((await call(agents,"advisor","GET","/api/lab/agents")).status).toBe(403);
  expect((await call(agents,"owner","GET","/api/lab/agents")).status).toBe(200);
  expect((await call(campaigns,"advisor","GET","/api/campaigns")).status).toBe(403);
  const result=await bodyOk(await call(campaigns,"owner","GET","/api/campaigns"));
  expect(result.campaigns.map((r:{id:string})=>r.id)).toEqual([campaignsIds[0]]);
  expect((await call(campaign,"owner","GET","/api/campaigns/"+campaignsIds[1],undefined,campaignsIds[1])).status).toBe(404);
  expect((await bodyOk(await call(writing,"advisor","GET","/api/writing-assist"))).available).toBe(false);
  expect((await call(suggest,"advisor","POST","/api/writing-assist",{text:"",action:"improve"})).status).toBe(422);
  const profile=await authWeb(new NextRequest(base+"/api/auth/get-session",{headers:native.advisor}));
  expect(profile.status===401 || (profile.status===200 && await profile.json()===null)).toBe(true);
});
